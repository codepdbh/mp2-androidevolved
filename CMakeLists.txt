cmake_minimum_required(VERSION 3.21)
project(MP2R LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mp2r_input STATIC src/input/touch_engine.cpp)
target_include_directories(mp2r_input PUBLIC src/input)
if(NOT ANDROID)
  enable_testing()
  add_executable(touch_engine_test tests/touch_engine_test.cpp)
  target_link_libraries(touch_engine_test PRIVATE mp2r_input)
  add_test(NAME touch_engine COMMAND touch_engine_test)
endif()
option(MP2R_INITIALIZATION_PROBE "Execute selected original InitSys functions" OFF)
if(MP2R_INITIALIZATION_PROBE)
  add_executable(initialization_probe tests/initialization_probe_main.cpp src/runtime/initialization_probe.cpp)
  target_include_directories(initialization_probe PRIVATE src/runtime private/generated/initialization-code private/generated/initialization
    external/PS2Recomp/ps2xRuntime/include external/PS2Recomp/ps2xRuntime/src/lib/Kernel
    external/PS2Recomp/ps2xIOP/include)
  if(ANDROID)
    target_include_directories(initialization_probe PRIVATE external/sse2neon)
    target_compile_definitions(initialization_probe PRIVATE USE_SSE2NEON)
  else()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME initialization_probe COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_initialization.py" "$<TARGET_FILE:initialization_probe>")
    add_executable(ee_semaphores_test tests/ee_semaphores_test.cpp)
    target_include_directories(ee_semaphores_test PRIVATE src/runtime)
    add_test(NAME ee_semaphores COMMAND ee_semaphores_test)
    add_executable(ee_timers_intc_test tests/ee_timers_intc_test.cpp)
    target_include_directories(ee_timers_intc_test PRIVATE src/runtime)
    add_test(NAME ee_timers_intc COMMAND ee_timers_intc_test)
    add_executable(ee_threads_test tests/ee_threads_test.cpp)
    target_include_directories(ee_threads_test PRIVATE src/runtime)
    add_test(NAME ee_threads COMMAND ee_threads_test)
    add_executable(ee_tlb_test tests/ee_tlb_test.cpp)
    target_include_directories(ee_tlb_test PRIVATE src/runtime)
    add_test(NAME ee_tlb COMMAND ee_tlb_test)
  endif()
endif()

option(MP2R_RESOURCE_PROBE "Load a private PS2 texture using librw without a window" OFF)
if(MP2R_RESOURCE_PROBE)
  enable_language(C)
  set(LIBRW_PLATFORM "NULL" CACHE STRING "Headless backend" FORCE)
  set(LIBRW_TOOLS OFF CACHE BOOL "" FORCE)
  set(LIBRW_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory(external/librw)
  add_executable(rw_texture_probe tests/rw_texture_probe.cpp)
  target_link_libraries(rw_texture_probe PRIVATE librw)
endif()
option(MP2R_RECOMP_PROBE "Compile and test two private PAL leaf translations" OFF)
if(MP2R_RECOMP_PROBE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" scripts/verify_probe_bytes.py
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" RESULT_VARIABLE probe_identity)
  if(NOT probe_identity EQUAL 0)
    message(FATAL_ERROR "PAL ELF identity/bytes verification failed")
  endif()
  set(recomp_dir "${CMAKE_CURRENT_SOURCE_DIR}/private/generated/pal")
  add_executable(recomp_leaf_probe tests/recomp_leaf_probe.cpp
    "${recomp_dir}/sub_0010DCC8_0x10dcc8.cpp"
    "${recomp_dir}/sub_001B1C20_0x1b1c20.cpp")
  target_include_directories(recomp_leaf_probe PRIVATE "${recomp_dir}"
    external/PS2Recomp/ps2xRuntime/include external/PS2Recomp/ps2xRuntime/src
    external/PS2Recomp/ps2xRuntime/src/lib/Kernel
    external/PS2Recomp/ps2xIOP/include)
  if(ANDROID)
    target_include_directories(recomp_leaf_probe PRIVATE external/sse2neon)
    target_compile_definitions(recomp_leaf_probe PRIVATE USE_SSE2NEON)
  endif()
  if(NOT ANDROID)
    add_test(NAME recomp_leaf_probe COMMAND recomp_leaf_probe)
  endif()
endif()
