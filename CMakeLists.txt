option(MP2R_GAME_PROGRAM "Compile every translated original function, one translation unit each" OFF)
set(MP2R_GAME_SAMPLE 0 CACHE STRING "Compile only the first N game functions (build-cost measurement)")
if(MP2R_GAME_PROGRAM)
  set(MP2R_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
  include(private/generated/full-glue/sources.cmake)
  if(MP2R_GAME_SAMPLE GREATER 0)
    list(SUBLIST MP2R_GAME_SOURCES 0 ${MP2R_GAME_SAMPLE} MP2R_GAME_SOURCES)
  endif()
  add_library(mp2r_game_code STATIC ${MP2R_GAME_SOURCES})
  if(NOT MP2R_GAME_SAMPLE GREATER 0)
    target_sources(mp2r_game_code PRIVATE private/generated/full-glue/register_full.cpp)
  endif()
  target_include_directories(mp2r_game_code PUBLIC src/runtime private/generated/full-glue
    external/PS2Recomp/ps2xRuntime/include external/PS2Recomp/ps2xRuntime/src/lib/Kernel
    external/PS2Recomp/ps2xIOP/include)
  target_precompile_headers(mp2r_game_code PRIVATE src/runtime/mp2_generated_prelude.h)
  if(MSVC)
    target_compile_options(mp2r_game_code PRIVATE /MP6 /bigobj /O1)
  endif()
  if(ANDROID)
    target_include_directories(mp2r_game_code PUBLIC external/sse2neon)
    target_compile_definitions(mp2r_game_code PUBLIC USE_SSE2NEON)
  endif()
  if(NOT MP2R_GAME_SAMPLE GREATER 0)
    add_executable(game_probe tests/game_probe_main.cpp src/runtime/initialization_probe.cpp src/runtime/mp2_guest_runtime.cpp)
    target_link_libraries(game_probe PRIVATE mp2r_game_code)
  endif()
endif()

option(MP2R_RECOMP_PROBE "Compile and test two private PAL leaf translations" OFF)
if(MP2R_RECOMP_PROBE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" scripts/verify_probe_bytes.py
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" RESULT_VARIABLE probe_identity)
  if(NOT probe_identity EQUAL 0)
    message(FATAL_ERROR "PAL ELF identity/bytes verification failed")
  endif()
  set(recomp_dir "${CMAKE_CURRENT_SOURCE_DIR}/private/generated/pal")
  add_executable(recomp_leaf_probe tests/recomp_leaf_probe.cpp
    "${recomp_dir}/sub_0010DCC8_0x10dcc8.cpp"
    "${recomp_dir}/sub_001B1C20_0x1b1c20.cpp")
  target_include_directories(recomp_leaf_probe PRIVATE "${recomp_dir}"
    external/PS2Recomp/ps2xRuntime/include external/PS2Recomp/ps2xRuntime/src
    external/PS2Recomp/ps2xRuntime/src/lib/Kernel
    external/PS2Recomp/ps2xIOP/include)
  if(ANDROID)
    target_include_directories(recomp_leaf_probe PRIVATE external/sse2neon)
    target_compile_definitions(recomp_leaf_probe PRIVATE USE_SSE2NEON)
  endif()
  if(NOT ANDROID)
    add_test(NAME recomp_leaf_probe COMMAND recomp_leaf_probe)
  endif()
endif()
